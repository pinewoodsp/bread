import pygame import sys import random # 1. 초기화 및 창 설정 pygame.init() screen_width = 800 screen_height = 600 screen = pygame.display.set_mode((screen_width, screen_height)) pygame.display.set_caption("나만의 귀여운 빵 만들기 미니게임") clock = pygame.time.Clock() # 2. 게임 상태(State) 정의 STATE_INGREDIENTS = 1 STATE_CHOOSE_SHAPE = 2 STATE_TIMING = 3 STATE_OVEN = 4 STATE_RESULT = 5 current_state = STATE_INGREDIENTS # 3. 게임 데이터 및 UI 영역(Rect) 설정 # 1단계 리소스 및 상태 ingredients_status = {"flour": False, "salt": False, "sugar": False, "yeast": False, "water": False} rect_flour = pygame.Rect(50, 50, 100, 100) rect_salt = pygame.Rect(170, 50, 100, 100) rect_sugar = pygame.Rect(50, 170, 100, 100) rect_yeast = pygame.Rect(170, 170, 100, 100) rect_water = pygame.Rect(50, 290, 100, 100) rect_next_btn = pygame.Rect(600, 500, 150, 50) # 2단계 선택 버튼 rect_baguette_btn = pygame.Rect(150, 400, 200, 80) rect_bread_btn = pygame.Rect(450, 400, 200, 80) selected_bread = None # 3단계 타이밍 게임 변수 timing_bar_x = 200 timing_direction = 1 timing_bar_speed = 7 target_zone_x = 400 target_zone_width = 80 timing_attempts = 3 timing_successes = 0 # 4단계 오븐 게임 변수 oven_gauge = 30 # 시작 게이지 온도 oven_timer = 300 # 60fps 기준 약 5초 제한시간 is_oven_heating = False # 5단계 결과 및 다시하기 버튼 result_state = 2 # 1: 덜구워짐, 2: 완벽, 3: 탐 rect_restart_btn = pygame.Rect(325, 520, 150, 50) # 4. 이미지 로드 try: img_bowl = pygame.image.load("bowl.jpg") img_flour = pygame.image.load("flour.jpg") img_salt = pygame.image.load("salt.png") img_sugar = pygame.image.load("sugar.jpg") img_yeast = pygame.image.load("yeast.png") img_water = pygame.image.load("water.jpg") img_note = pygame.image.load("note.jpg") img_kneading = pygame.image.load("kneading.png") img_rolling_pin = pygame.image.load("rolling_pin.png") img_oven_off = pygame.image.load("oven_off.jpg") img_oven_on = pygame.image.load("oven_on.jpg") results = { "baguette": { 1: pygame.image.load("baguette1.png"), 2: pygame.image.load("baguette2.jpg"), 3: pygame.image.load("baguette3.png") }, "bread": { 1: pygame.image.load("bread1.jpg"), 2: pygame.image.load("bread2.jpg"), 3: pygame.image.load("bread3.jpg") } } except pygame.error as e: print(f"이미지 로드 에러 (파일명이나 경로를 확인하세요): {e}") sys.exit() # 5. 폰트 설정 font = pygame.font.SysFont("malgungothic", 24) font_large = pygame.font.SysFont("malgungothic", 32) # 6. 메인 게임 루프 running = True while running: dt = clock.tick(60) screen.fill((250, 245, 235)) # 따뜻한 베이지색 배경 # ---- [이벤트 및 조작 처리] ---- for event in pygame.event.get(): if event.type == pygame.QUIT: running = False elif event.type == pygame.MOUSEBUTTONDOWN: mouse_pos = event.pos # 1단계 크릭 이벤트 if current_state == STATE_INGREDIENTS: if rect_flour.collidepoint(mouse_pos): ingredients_status["flour"] = True elif rect_salt.collidepoint(mouse_pos): ingredients_status["salt"] = True elif rect_sugar.collidepoint(mouse_pos): ingredients_status["sugar"] = True elif rect_yeast.collidepoint(mouse_pos): ingredients_status["yeast"] = True elif rect_water.collidepoint(mouse_pos): ingredients_status["water"] = True # 모든 재료가 다 들어갔을 때 '다음' 버튼 클릭 if all(ingredients_status.values()) and rect_next_btn.collidepoint(mouse_pos): current_state = STATE_CHOOSE_SHAPE # 2단계 모양 선택 이벤트 elif current_state == STATE_CHOOSE_SHAPE: if rect_baguette_btn.collidepoint(mouse_pos): selected_bread = "baguette" target_zone_x = random.randint(250, 480) # 판정존 랜덤 생성 current_state = STATE_TIMING elif rect_bread_btn.collidepoint(mouse_pos): selected_bread = "bread" target_zone_x = random.randint(250, 480) current_state = STATE_TIMING # 3단계 타이밍 맞추기 이벤트 elif current_state == STATE_TIMING: # 노란 바가 녹색 판정 범위 안에 들어왔는지 검사 if target_zone_x <= timing_bar_x <= (target_zone_x + target_zone_width): timing_successes += 1 timing_attempts -= 1 target_zone_x = random.randint(250, 480) # 성공/실패 후 판정존 재배치 if timing_attempts <= 0: # 타이밍 성형 성공 횟수에 따라 오븐 보너스 시작 온도 지급 oven_gauge = 20 + (timing_successes * 10) current_state = STATE_OVEN # 4단계 오븐 연타 이벤트 (마우스 클릭) elif current_state == STATE_OVEN: oven_gauge += 4 is_oven_heating = True # 5단계 다시하기 버튼 elif current_state == STATE_RESULT: if rect_restart_btn.collidepoint(mouse_pos): # 모든 데이터 리셋 ingredients_status = {k: False for k in ingredients_status} timing_attempts = 3 timing_successes = 0 oven_timer = 300 current_state = STATE_INGREDIENTS elif event.type == pygame.KEYDOWN: # 4단계 오븐 스페이스바 연타 기능 지원 if event.key == pygame.K_SPACE and current_state == STATE_OVEN: oven_gauge += 4 is_oven_heating = True # ---- [상태별 화면 렌더링 및 실시간 로직] ---- # 1단계: 재료 넣기 화면 if current_state == STATE_INGREDIENTS: # 가이드 가죽 메모장 screen.blit(img_note, (520, 50)) recipe_txt = font.render("레시피: 전부 다 넣기!", True, (50, 30, 10)) screen.blit(recipe_txt, (550, 120)) # 재료 아이콘 및 상태 텍스트 materials = [(img_flour, rect_flour, "밀가루"), (img_salt, rect_salt, "소금"), (img_sugar, rect_sugar, "설탕"), (img_yeast, rect_yeast, "이스트"), (img_water, rect_water, "물")] for img, rect, name in materials: screen.blit(img, (rect.x, rect.y)) # 넣은 재료는 체크 표시 if ingredients_status[name.replace("밀가루","flour").replace("소금","salt").replace("설탕","sugar").replace("이스트","yeast").replace("물","water")]: pygame.draw.rect(screen, (0, 200, 0), rect, 4) screen.blit(img_bowl, (280, 200)) # 다 넣으면 다음 단계 버튼 활성화 if all(ingredients_status.values()): pygame.draw.rect(screen, (255, 150, 50), rect_next_btn, border_radius=10) next_txt = font.render("구우러 가기 ->", True, (255, 255, 255)) screen.blit(next_txt, (rect_next_btn.x + 15, rect_next_btn.y + 12)) # 2단계: 빵 모양 선택 화면 (중간 추가 기획 변경지점) elif current_state == STATE_CHOOSE_SHAPE: screen.blit(img_kneading, (320, 150)) screen.blit(img_rolling_pin, (120, 220)) pygame.draw.rect(screen, (235, 175, 110), rect_baguette_btn, border_radius=10) pygame.draw.rect(screen, (210, 140, 80), rect_bread_btn, border_radius=10) txt1 = font.render("바게트 모양", True, (255, 255, 255)) txt2 = font.render("식빵 모양", True, (255, 255, 255)) screen.blit(txt1, (rect_baguette_btn.x + 40, rect_baguette_btn.y + 25)) screen.blit(txt2, (rect_bread_btn.x + 50, rect_bread_btn.y + 25)) title_txt = font_large.render("어떤 모양의 빵을 만드시겠습니까?", True, (60, 40, 20)) screen.blit(title_txt, (180, 50)) # 3단계: 타이밍 성형 미니게임 elif current_state == STATE_TIMING: # 타이밍 바 이동 로직 timing_bar_x += timing_bar_speed * timing_direction if timing_bar_x <= 200 or timing_bar_x >= 600: timing_direction *= -1 # 튕기기 # UI 드로잉 title_txt = font_large.render("밀대로 반죽 밀기! (타이밍 맞춰 클릭)", True, (0, 0, 0)) screen.blit(title_txt, (160, 80)) # 게이지 전체 트랙 (레일) pygame.draw.rect(screen, (200, 200, 200), [200, 300, 400, 40], border_radius=5) # 초록색 정답 타겟 구역 pygame.draw.rect(screen, (50, 220, 50), [target_zone_x, 300, target_zone_width, 40]) # 움직이는 노란색 유저 선 pygame.draw.rect(screen, (255, 215, 0), [timing_bar_x, 290, 8, 60]) info_txt = font.render(f"남은 기회: {timing_attempts}번 | 성공: {timing_successes}번", True, (50, 50, 50)) screen.blit(info_txt, (300, 420)) # 4단계: 오븐 굽기 (연타 미니게임) elif current_state == STATE_OVEN: oven_timer -= 1 # 자연 냉각 (온도가 서서히 떨어짐) if oven_gauge > 0: oven_gauge -= 0.25 oven_gauge = max(0, min(100, oven_gauge)) # 0~100 제한 # 애니메이션 효과 (연타 중일 땐 잠깐 ON 이미지 불 켜지게 함) if is_oven_heating: screen.blit(img_oven_on, (260, 80)) is_oven_heating = False # 다음 프레임에 꺼지도록 초기화 else: screen.blit(img_oven_off, (260, 80)) # 온도 게이지 바 UI 구현 (0~40 블루존 / 40~75 스위트 세이프 레드존 / 75~100 경고존) pygame.draw.rect(screen, (220, 220, 220), [200, 440, 400, 30], border_radius=5) # 베이스 # 현재 온도 게이지 bar_width = int(oven_gauge * 4) gauge_color = (100, 150, 255) if oven_gauge < 40 else ((50, 200, 50) if oven_gauge <= 75 else (255, 50, 50)) pygame.draw.rect(screen, gauge_color, [200, 440, bar_width, 30], border_radius=5) # 완벽한 온도 가이드 경계선 표시 pygame.draw.line(screen, (0,0,0), (200 + 160, 440), (200 + 160, 470), 2) # 40%선 pygame.draw.line(screen, (0,0,0), (200 + 300, 440), (200 + 300, 470), 2) # 75%선 txt_guide = font.render("← 덜구워짐 [ 완벽함 존! ✨ ] 타버림 →", True, (80, 80, 80)) screen.blit(txt_guide, (230, 480)) timer_txt = font_large.render(f"남은 시간: {max(0, oven_timer // 60)}초", True, (255, 0, 0)) screen.blit(timer_txt, (320, 20)) # 시간 종료 시 결과 판정 로직 if oven_timer <= 0: if oven_gauge < 40: result_state = 1 # 덜 구워짐 elif oven_gauge <= 75: result_state = 2 # 완벽함 else: result_state = 3 # 타버림 current_state = STATE_RESULT # 5단계: 최종 결과 화면 elif current_state == STATE_RESULT: final_image = results[selected_bread][result_state] screen.blit(final_image, (260, 120)) status_text = { 1: "앗... 설익어서 밀가루 맛이 나요 😭 (덜 구워짐)", 2: "겉바속촉 완벽하게 구워진 최고의 명작! ✨ (성공)", 3: "오븐에 불이 붙을 뻔했습니다 흑흑 🖤 (타버림)" } res_txt = font_large.render(status_text[result_state], True, (40, 20, 10)) screen.blit(res_txt, (100, 430)) # 다시하기 버튼 pygame.draw.rect(screen, (100, 100, 100), rect_restart_btn, border_radius=8) restart_txt = font.render("다시 만들기", True, (255, 255, 255)) screen.blit(restart_txt, (rect_restart_btn.x + 23, rect_restart_btn.y + 12)) pygame.display.flip() pygame.quit() sys.exit()
